    <p>اضغط هنا لرفع الصورة أو اسحبها إلى هنا</p>

    <label for="quality">جودة الضغط: <span id="qualityVal">80</span>%</label>
    <input type="range" id="quality" min="10" max="100" value="80">

    <div class="preview-container">
      <div class="preview-box">
        <h4>الصورة الأصلية</h4>
        <img id="originalImg" src="" alt="Original">
        <p id="originalSize"></p>
      </div>
      <div class="preview-box">
        <h4>بعد الضغط</h4>
        <img id="compressedImg" src="" alt="Compressed">
        <p id="compressedSize"></p>
      </div>
    </div>

    <a id="downloadBtn" download="compressed-image.jpg" style="text-decoration: none;">
      <button class="btn">تحميل الصورة المضغوطة</button>
    </a>

<script>
  const imageInput = document.getElementById('imageInput');
  const qualityInput = document.getElementById('quality');
  const qualityVal = document.getElementById('qualityVal');
  const originalImg = document.getElementById('originalImg');
  const compressedImg = document.getElementById('compressedImg');
  const originalSize = document.getElementById('originalSize');
  const compressedSize = document.getElementById('compressedSize');
  const downloadBtn = document.getElementById('downloadBtn');
  const controls = document.getElementById('controls');

  let fileOriginal = null;

  imageInput.addEventListener('change', (e) => {
    fileOriginal = e.target.files[0];
    if (!fileOriginal) return;

    originalSize.innerText = (fileOriginal.size / 1024).toFixed(2) + " KB";
    const reader = new FileReader();
    reader.onload = (event) => {
      originalImg.src = event.target.result;
      controls.style.display = 'block';
      compressImage();
    };
    reader.readAsDataURL(fileOriginal);
  });

  qualityInput.addEventListener('input', () => {
    qualityVal.innerText = qualityInput.value;
    compressImage();
  });

  function compressImage() {
    const canvas = document.createElement('canvas');
    const ctx = canvas.getContext('2d');
    const img = new Image();
    img.src = originalImg.src;

    img.onload = () => {
      canvas.width = img.width;
      canvas.height = img.height;
      ctx.drawImage(img, 0, 0);

      const quality = qualityInput.value / 100;
      const dataUrl = canvas.toDataURL('image/jpeg', quality);
      compressedImg.src = dataUrl;
      downloadBtn.href = dataUrl;

      // حساب الحجم التقريبي للـ Base64
      const head = 'data:image/jpeg;base64,';
      const sizeInBytes = Math.round((dataUrl.length - head.length) * 3 / 4);
      compressedSize.innerText = (sizeInBytes / 1024).toFixed(2) + " KB";
    };
  }
</script>
